        function handlePhotoUpload(event) {

            const file = event.target.files[0];

            if (!file) return;

            const reader = new FileReader();

            reader.onload = function(e) {

                const img = new Image();

                img.onload = function() {

                    const MAX_WIDTH = 800;

                    let width = img.width;

                    let height = img.height;

                    if (width > MAX_WIDTH) {

                        height = Math.round((height * MAX_WIDTH) / width);

                        width = MAX_WIDTH;

                    }

                    const canvas = document.createElement('canvas');

                    canvas.width = width;

                    canvas.height = height;

                    const ctx = canvas.getContext('2d');

                    ctx.drawImage(img, 0, 0, width, height);

                    currentPhotoBase64 = canvas.toDataURL('image/jpeg', 0.75);

                    showPhotoPreview(currentPhotoBase64);

                };

                img.src = e.target.result;

            };

            reader.readAsDataURL(file);

        }

        async function saveRecipe() {

            if (!title) {

                showToast('Введите название рецепта');

                const name = row.querySelector('.ing-name').value.trim();

                const amount = row.querySelector('.ing-amount').value.trim();

                const unit = row.querySelector('.ing-unit').value;

            }

            const stepsRaw = document.getElementById('form-steps').value.trim();

            const steps = stepsRaw ? stepsRaw.split('\n').map(s => s.trim()).filter(Boolean) : [];

            const existing = allRecipes.find(r => r.id === id);

                title: title,

                category: document.getElementById('form-category').value.trim() || 'Основные',

                time: document.getElementById('form-time').value.trim(),

                image: currentPhotoBase64,

                ingredients: ingredients,

                steps: steps,

                notes: document.getElementById('form-notes').value.trim(),

                createdAt: existing ? existing.createdAt : Date.now(),

                updatedAt: Date.now()

            };

                await RecipeDB.save(recipeObj);

        function importRecipes(event) {

            const file = event.target.files[0];

            if (!file) return;

            const reader = new FileReader();

            reader.onload = async function(e) {

                try {

                    const imported = JSON.parse(e.target.result);

                    if (!Array.isArray(imported)) {

                        throw new Error('Неверный формат файла');

                    }

                   

                    const map = new Map();

                    allRecipes.forEach(r => map.set(r.id, r));

                    imported.forEach(r => {

                        if (r.id && r.title) map.set(r.id, r);

                    });

                    const merged = Array.from(map.values());

                    await RecipeDB.clearAndBulkInsert(merged);

                    allRecipes = await RecipeDB.getAll();

                   

                    closeSettingsModal();

                    showHomeView();

                    showToast(`Загружено рецептов: ${imported.length}`);

                } catch (err) {

                    showToast('Ошибка: поврежденный файл копии');

                    console.error(err);

                } finally {

                    event.target.value = '';

                }

            };

            reader.readAsText(file);

        }

        function pluralizeIngredients(count) {

            const mod10 = count % 10;

            const mod100 = count % 100;

            if (mod10 === 1 && mod100 !== 11) return 'ингредиент';

            if (mod10 >= 2 && mod10 <= 4 && (mod100 < 10 || mod100 >= 20)) return 'ингредиента';

            return 'ингредиентов';

        }
